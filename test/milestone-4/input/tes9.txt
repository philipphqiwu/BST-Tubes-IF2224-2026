program TestFunc;

var
  n, result: integer;

function Factorial(k: integer): integer;
begin
  if k <= 1 then
    Factorial := 1
  else
    Factorial := k * Factorial(k - 1);
end;

function Square(x: integer): integer;
begin
  Square := x * x;
end;

begin
  n := 5;
  result := Factorial(n);
  writeln('fact(', n, ') = ', result);
  writeln('square(', n, ') = ', Square(n));
end.
