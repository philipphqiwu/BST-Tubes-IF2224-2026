program ControlFlow;

var
  i, x, y: integer;

begin
  x := 0;
  for i := 1 to 3 do
    begin
      x := x + i;
      y := 1;
    end;

  while x > 0 do
    begin 
      x := x - 1;
      y := 0;
    end;

  repeat
    x := x + 2;
  until x > 1;

  if x > 1 then
    x := 5
  else
    x := 6;

  case x of
    1: x := 10;
    2, 3: x := 20;
  end;
end.
